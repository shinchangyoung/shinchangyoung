# @title 🎯 구글 드라이브 마운트 및 경로 확인
from google.colab import drive
import os

# 1. 구글 드라이브 마운트
drive.mount('/content/drive')

# 2. 경로 설정 (Windows의 "G:\내 드라이브\shin"은 Colab에서 아래와 같습니다)
model_folder = "/content/drive/MyDrive/shin"
model_name = "sd_xl_base_1.0.safetensors"
model_path = os.path.join(model_folder, model_name)

# 3. 폴더 탐색 및 파일 확인
print(f"📁 폴더 탐색: {model_folder}")

if os.path.exists(model_folder):
    print("✅ 폴더가 존재합니다. 파일 목록을 출력합니다:")
    !ls -la "{model_folder}"
    
    # 4. 특정 모델 파일 존재 여부 확인
    if os.path.exists(model_path):
        size_gb = os.path.getsize(model_path) / (1024**3)
        print(f"\n✨ 모델 파일 발견!")
        print(f"📍 경로: {model_path}")
        print(f"📏 크기: {size_gb:.2f} GB")
    else:
        print(f"\n❌ 파일을 찾을 수 없습니다: {model_name}")
        print("💡 대소문자나 파일 확장자를 다시 확인해주세요.")
else:
    print(f"❌ 폴더를 찾을 수 없습니다. 마운트가 정상적으로 되었는지 확인해주세요.")
